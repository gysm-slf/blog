const SSEDatasSplit = function (datas = '') {

const parseDataByJSONPath = function (nodes = [], JSONPath) {
  if (JSONPath === '') {
    return nodes.join('') // 返回 answer 内容
  } else {
    nodes = nodes.map(node => {
      const parse = jsonProcessor.parse(node)
      if (typeof parse === 'undefined') {
        console.warn('无法解析的节点 --> ', node)
      }
      return parse
    })
    if (JSONPath === '$') { // 返回可解析的事件节点对象数组
      return nodes
    } else { // 返回 answer 内容
      return nodes.map(node => {
        let result
        try {
          result = jp.query(node, JSONPath)
        } catch (error) {
          console.error(error.message)
          console.warn(`读取节点消息失败，JSONPath：${JSONPath}，节点内容：${node}`)
        }
        return result || ''
      }).join('')
    }
  }
}

   * 
   * @param response 调用 fetch 后直接传入即可 
   * @param onReceived 解析回调函数 参数为每次读取的内容
   * @param onFinally
   * @param onFailed
   * @param nodeSplit 节点拆分函数 回调参数为读取的字符串 要求返回分割后的数组
   * @param JSONPath ''|'$'|'$.xxx'(可参考 jsonpath_ng 库语法，例如 OpenAI 类接口为：$.choices[0].delta.content)
   * ''|'$.xxx': 将返回内容视作消息直接返回 '$': 将返回内容视作 json 格式对象解析为对象后直接返回 
   * xml 格式流接口也可以使用，设置为空后内容会原样返回，然后在 onReceived 回调中自行提取即可
   */
  handleStream: function (
    response, 
    { 
      onReceived = $onReceived,
      onComplete = $onComplete,
      onError = $onError,
      nodeSplit = SSEDatasSplit,
      JSONPath = '$',
      abortController // 如果需要支持中断可传入请求时的 AbortController 实例
    } = {}
  ) {
    const reader = response.body.getReader()
    const received = { buffer: '' }
    const readData = async () => {
      try {
        while (true) {
          const { value, done } = await reader.read({ signal: abortController?.signal })
          received.buffer += new TextDecoder().decode(value, { stream: true })
          const nodes = nodeSplit(received.buffer) // 字符数组
          received.buffer = nodes.pop()
          onReceived(parseDataByJSONPath(nodes, JSONPath))
          if (done) {
            break
          }
        }
      } catch (error) {
        if (error.name === 'AbortError') {
          console.log('请求被中止')
        } else {
          console.error('Error reading stream:', error)
        }
        onError(error)
      } finally {
        if (received.buffer) { // 如果存在剩余的缓存未处理则最后一次进行接收回调
          onReceived(parseDataByJSONPath([received.buffer], JSONPath))
        }
        try {
          if (reader && !reader.closed) {
            reader.releaseLock();
          }
        } catch (e) {
          console.warn('释放 reader 锁失败:', e);
        }
        onComplete()
      }
    }
    readData()
  }
}

/**
 * 使用方法：
 * 1. 在 fetch 调用前调用
 *   softAutoScroll.switchInit() // 初始化柔和滚动工具开关，如果不需要每次提问后自动滚动到最底部而是继承柔和滚动开关则可删除该行
 * 2. 在 onReceived 回调中调用（可视情况决定是否使用 nextTick）  
 *   softAutoScroll.update()
 */
const SOFT_AUTO_SCROLL_INSTANCE = {
  container: null, // 暂时不考虑一屏幕内存在多自动滚动情况，如有需求在进行扩展
  scrollSwitch: true, // 
  balance: 10,
  /**
   * 自适应滚动
   * @param selector
   * @param balance 平衡值：取值越大时触发自动滚动区域越大
   */
  enable (selector, balance = 10) {
    if (selector) {
      this.container = document.querySelector(selector)
      this.container.removeEventListener('scroll', this.handleSoftAutoScroll)
      this.container.addEventListener('scroll', this.handleSoftAutoScroll)
      this.balance = balance
    }
  },
  disable () {
    this.scrollSwitch = false
    if (this.container) {
      this.container.removeEventListener('scroll', this.handleSoftAutoScroll)
      this.container = null
    }
  },
  isAtBottom: function () {
    if (this.container) {
      return this.container.scrollHeight - this.container.clientHeight <= this.container.scrollTop + this.balance
    }
  },
  handleSoftAutoScroll: function () {
    if (this.scrollSwitch && !this.isAtBottom()) {
      this.scrollSwitch = false
    } else if (!this.scrollSwitch && this.isAtBottom()) {
      this.scrollSwitch = true
    }
  },
  update: function () {
    if (this.container && this.scrollSwitch) {
      this.container.scrollTop = this.container.scrollHeight
    }
  },
  switchInit () {
    if (this.container) {
      this.scrollSwitch = true
    }
  },
}
export default {
  enable: function (...args) {
    SOFT_AUTO_SCROLL_INSTANCE.enable(...args)
  },
  disable: function () {
    SOFT_AUTO_SCROLL_INSTANCE.disable()
  },
  /**
   * 当用户向上浏览时开关会被关闭，若此时重新提问开关依然是关闭的状态，会导致该次发送消息时未滚动到最新发送消息处
   * 若想开启柔和滚动后，且每次发送消息后滚动到最新发送消息处时在流接口调用前调用 init 方法即可
   */
  switchInit: function () {
    SOFT_AUTO_SCROLL_INSTANCE.switchInit()
  },
  update: function () {
    SOFT_AUTO_SCROLL_INSTANCE.update()
  }
}

  async createImageRun (imgEl) {
    const src = imgEl.src || imgEl.getAttribute('src');
    if (!src) return null;

    let buffer;
    let contentType = 'image/png';

    // 1. 加载图片数据（保持不变）
    if (src.startsWith('data:')) {
      const matches = src.match(/^data:(image\/[a-zA-Z+]+);base64,(.+)$/);
      if (!matches) return null;
      contentType = matches[1];
      const base64Data = matches[2];
      const binaryString = atob(base64Data);
      const len = binaryString.length;
      const bytes = new Uint8Array(len);
      for (let i = 0; i < len; i++) bytes[i] = binaryString.charCodeAt(i);
      buffer = bytes;
    } else {
      const response = await fetch(src);
      if (!response.ok) throw new Error(`图片加载失败: ${src}`);
      const arrayBuffer = await response.arrayBuffer();
      buffer = new Uint8Array(arrayBuffer);
      const ct = response.headers.get('content-type');
      if (ct && ct.startsWith('image/')) contentType = ct;
    }

    // --- 2. 修复：安全的尺寸计算逻辑 ---
    const MAX_WIDTH_PX = 600; // A4纸页边距内通常约 6-700 像素

    const DEFAULT_HEIGHT_PX = 200;

    let width = DEFAULT_WIDTH_PX;
    let height = DEFAULT_HEIGHT_PX;

  async createTableFromElement (tableEl, inheritedTextOptions = {}, parentBlockParaProps = {}) {
    const rows = [];

    // 1. 安全获取表格行（兼容 thead/tbody/tfoot）
    const getRowsFromContainer = (container) => {
      return Array.from(container.children).filter(child => child.tagName.toUpperCase() === 'TR');

    rowElements.push(...getRowsFromContainer(tableEl)); // 直接子 TR

      sections.forEach(sec => rowElements.push(...getRowsFromContainer(sec)));

    if (rowElements.length === 0) return null;

    // 2. 遍历行
    for (const tr of rowElements) {

        ['TD', 'TH'].includes(child.tagName.toUpperCase())
      );

        const cellBlocks = [];
        // 用于临时存放行内内容（TextRun），凑够一波就打包成 Paragraph

              // 1. 先把之前攒的行内内容打包成段落

                  ...parentBlockParaProps,

                }));

            } else if (item instanceof TextRun) {
              // 如果是行内文本：暂时攒起来

            }
          }
        }

        // 3.2 循环结束后，把最后攒的行内内容也打包成段落

            ...parentBlockParaProps,
            children: currentInlineRuns
          }));

  // --- 核心：递归扁平化 DOM 为段落/图片数组 ---
  async processNode(node, inheritedTextOptions = {}, parentBlockParaProps = {}) {
    const results = [];

    // 文本节点
    if (node.nodeType === Node.TEXT_NODE) {
      const text = node.textContent;
      if (text.trim() === '') {
        // 保留空格（作为 TextRun）
        results.push(new TextRun({ text, ...inheritedTextOptions }));
      } else {
        results.push(new TextRun({ text, ...inheritedTextOptions }));
      }
      return results;
    }

    // 非元素节点直接返回空
    if (node.nodeType !== Node.ELEMENT_NODE) return results;

    const el = node;
    const tag = el.tagName.toUpperCase();

    // 处理图片
    if (tag === 'IMG') {
      try {
        const imgRun = await this.createImageRun(el);
        if (imgRun) {
          // 图片始终作为一个独立段落返回
          results.push(new Paragraph({ children: [imgRun] }));
        }
      } catch (e) {
        console.warn('图片处理失败:', e);
      }
      return results;
    }

    // --- 新增：处理表格 ---
    if (tag === 'TABLE') {
      try {
        // 注意：这里传入当前的文本选项和段落属性
        const table = await this.createTableFromElement(el, inheritedTextOptions, parentBlockParaProps);
        if (table) {
          // 表格在 docx 中是顶级元素，直接 push
          results.push(table);
        }
      } catch (e) {
        console.error('表格处理失败:', e); // 改为 console.error 看详细堆栈
      }
      return results;
    }

    // 处理换行符
    if (tag === 'BR') {
      results.push(new TextRun({ break: 1 }));
      return results;
    }

    // 判断当前元素是否为块级
    const isBlock = this.isBlockElement(el);

    if (isBlock) {
      // ----- 块级元素：生成一个或多个段落 -----
      // 获取该元素自己的段落属性（合并父块级的段落属性？通常段落属性不从父级继承，但缩进等可能需要叠加）
      const ownParaProps = this.getParagraphProperties(el);
      // 合并父块级段落属性（如 margin 叠加？简单起见，直接用自身的，但 left indent 可能需要累加？视情况而定）
      const mergedParaProps = { ...parentBlockParaProps, ...ownParaProps };

      // 计算当前元素内文本的基础样式（继承父样式 + 自身样式）
      const currentTextOptions = this.getTextRunOptionsFromElement(el, inheritedTextOptions);

      // 收集该块级元素内部的所有子节点，将它们转换为段落序列或行内运行
      const childNodes = Array.from(el.childNodes);
      const inlineRuns = [];          // 当前段落累积的行内内容
      let lastWasBlockChild = false;  // 上一个子节点是否为块级

      for (const child of childNodes) {
        // 如果是块级子元素，需要将之前累积的行内内容输出为一个段落，然后递归处理该子元素
        if (child.nodeType === Node.ELEMENT_NODE && this.isBlockElement(child)) {
          // 先输出累积的行内内容
          if (inlineRuns.length > 0) {
            results.push(new Paragraph({
              ...mergedParaProps,
              children: inlineRuns.splice(0, inlineRuns.length)
            }));
          }
          // 递归处理块级子元素（传递当前块级的段落属性作为父级？若需要嵌套缩进叠加可传递 mergedParaProps）
          const childResults = await this.processNode(child, currentTextOptions, mergedParaProps);
          results.push(...childResults);
          lastWasBlockChild = true;
        } else {
          // 行内内容（文本、行内元素、BR、IMG等）
          const childResults = await this.processNode(child, currentTextOptions, mergedParaProps);
          // 将返回的内容添加到 inlineRuns 中
          for (const item of childResults) {
            if (item instanceof Paragraph) {
              // 如果 processNode 返回了段落（例如图片），则先输出累积的行内，再输出该段落
              if (inlineRuns.length > 0) {
                results.push(new Paragraph({ ...mergedParaProps, children: inlineRuns.splice(0, inlineRuns.length) }));
              }
              results.push(item);
              lastWasBlockChild = true;
            } else if (item instanceof TextRun) {
              inlineRuns.push(item);
              lastWasBlockChild = false;
            }
          }
        }
      }

      // 循环结束，输出剩余的行内内容
      if (inlineRuns.length > 0) {
        results.push(new Paragraph({
          ...mergedParaProps,
          children: inlineRuns
        }));
      }

      // 如果该块级元素没有任何内容（空段落），也保留一个空段落以维持间距（可选）
      if (results.length === 0 && el.innerText.trim() === '' && el.querySelectorAll('img').length === 0) {
        // 可忽略
      }

      return results;
    } else {
      // ----- 行内元素：直接将其内容转换为 TextRun 数组，并叠加样式 -----
      const currentOptions = this.getTextRunOptionsFromElement(el, inheritedTextOptions);
      const childResults = [];
      for (const child of el.childNodes) {
        const subResults = await this.processNode(child, currentOptions, parentBlockParaProps);
        childResults.push(...subResults);
      }
      return childResults; // 应该全部是 TextRun 或 换行符
    }
  },

  // --- 入口：转换整个容器 ---
  async convertElementToDocx(element) {
    const rawChildren = [];
    const childNodes = Array.from(element.childNodes);
    for (const child of childNodes) {
      const results = await this.processNode(child, {}, {});
      rawChildren.push(...results);
    }

    // 将所有裸露的 TextRun 收集成段落，确保只有块级元素进入 section
    const children = [];
    let inlineBuffer = [];
    for (const item of rawChildren) {
      if (item instanceof Paragraph || item instanceof Table) {
        if (inlineBuffer.length > 0) {
          children.push(new Paragraph({ children: inlineBuffer.splice(0) }));
        }
        children.push(item);
      } else if (item instanceof TextRun) {
        inlineBuffer.push(item);
      }
    }
    if (inlineBuffer.length > 0) {
      children.push(new Paragraph({ children: inlineBuffer }));
    }
    return children;
  }
}

## 事件管理

```js
/**
 * 功能描述:事件管理中心
 * 对比Vue.$emit特点：
 * 1.支持单一事件接收返回值
 * 2.支持同一类型注册多事件，且支持按优先级顺序执行
 * 3.可跨任意层级
 * 4.支持一次性事件
 * 5.支持在某条件下取消注册的事件（需要保留注册函数的引用）
 * 6.支持事件类型独占（isRefresh：true 以最后一次注册为准）
 * 7.支持链式注册与移除
 *
 * ！！！ 值得注意的一点是，要注意组件的生命周期，要保证事件先注册，后触发，否则第一次事件将无法正常触发
 * ！！！ 值得注意的一点是，如果不熟悉该组件建议注册事件时第三个参数固定传true，避免重复注册 （或者在组件销毁时移除相应事件）
 * @author: slf
 * @date: 2024/12/19 16:09
 */

export const EventManagementCenter = (function () {
  const events = {}
  return {
    // type: 事件名称 fn: 事件处理器  isRefresh:是否刷新(注册事件时会把之前注册过的同名事件先清空,如果出现组件内切换导致重复注册了事件可打开此选项)
    regist: function (type, fn, isRefresh = false, { priority = 0 } = {}) { // 订阅某个事件
      fn = typeof fn === 'function' ? fn : () => {}
      fn.priority = priority
      if (typeof events[type] === 'undefined' || isRefresh) {
        events[type] = [fn]
      } else {
        if (events[type].map(f => jsonProcessor.stringify(f)).indexOf(jsonProcessor.stringify(fn)) === -1) {
          events[type].push(fn)
        }
      }
      return this
    },
    emit: function (type) { // 向订阅者发送信息
      if (!events[type] || events[type].length === 0) return
      if (events[type].length === 1) return events[type][0].call(this, ...(Array.from(arguments).slice(1)))
      events[type].sort((a, b) => b.priority - a.priority)
      let cacheLen = events[type].length
      for (let i = 0; i < events[type].length; i++) { // 注意，当某事件需要多处订阅时，不支持返回注册的函数结果 因为无法保证与识别对应顺序 也不要调整存储结构增加复杂性 一般不建议使用emit的返回值 即使是支持该功能的时候 如果需要接收多返回值 建议使用不同事件名称，最终效果是一致的
        events[type][i].call(this, ...(Array.from(arguments).slice(1)))
        if (cacheLen > events[type].length) { // 发生触发后即时销毁事件(即在一个类型事件调用期间,发生了部分事件取消订阅) 依旧从原索引处任务开始
          i--
          cacheLen = events[type].length
        }
      }
    },
    remove: function (type, fn) {
      if (typeof fn === 'function') {
        for (let i = events[type].length - 1; i >= 0; i--) {
          if (events[type][i] === fn) events[type].splice(i, 1)
        }
      } else {
        delete events[type]
      }
      return this
    }
  }
})()
```

::: warning 注意：
当在 `vue` 组件中使用时，要考虑几点：

1. 组件中使用 `emit`：如果可能存在多个并行子组件实例同时运行，当某组件触发事件后，需要在回调函数中通过参数区分触发源。
2. 组件中使用 `regist`：会导致重复注册同一事件，要考虑是否会对业务有影响，一般不建议如此"反向"调用。
:::

export default {
  untilExecute: function (fn, { ms = 100, maxTimeMs = 5000, errorHandler = msg => console.warn(msg) } = {}) {
    const startTime = Date.now()
    const retry = () => {
      if (Date.now() - startTime >= maxTimeMs) {
        console.log(fn)
        errorHandler(new Error(`After ${maxTimeMs}ms, the function fails to execute correctly`))
        return
      }
      const result = fn() // 因遵循 Promises/A+ 规范，此处可以将fn调用使用promise封装起来，这样就无需额外判断返回值类型
      if (result instanceof Promise) {
        result.then(status => {
          if (!status) {
            setTimeout(() => retry(), ms)
          }
        })
      } else {
        if (!result) {
          setTimeout(() => retry(), ms) // 减少性能开销
        }
      }
    }
    retry()
  },
  reduceFrequency: (event, { key = 'default__', delay_times = 200 } = {}, eventArgs = []) => {
    this['time__out__' + key] !== null && clearTimeout(this['time__out__' + key])
    this['time__out__' + key] = setTimeout(() => {
      if (!(eventArgs instanceof Array)) {
        eventArgs = [eventArgs]
      }
      event(...eventArgs)
      this['time__out__' + key] = null
    }, delay_times)
  }
}

export const cachedRequest = function (key = '', promise = () => new Promise(), {
  isRefresh = false,
  requestErrorIsRetry = false,
  requestRetryMaxTimes = 3, // 期望值  极端情况下可能会出现次数不足的情况
  retryInterval = 3000
} = {}) {
  if (isRefresh) {
    requestCache.remove(key)
    requestPromiseCache.remove(key)
  }
  return new Promise((resolve, reject) => {
    let retryTimes = 0
    untilExecute(() => {
      if (requestCache.has(key)) {
        resolve(requestCache.get(key))
        return true
      } else if (!(requestPromiseCache.has(key))) {
        return new Promise(innerResolve => { // untilExecute promise
          requestPromiseCache.set(key, promise().then((response = {}) => { // promise函数若执行了.catch请return { type: 'error', msg: '' }
            if (response.type !== 'error') {
              resolve(requestCache.set(key, response)) // 成功后untilExecute不再执行下一次回调 因此此处直接返回 或者直接 innerResolve(false)也可以 但是这样效率更高
              innerResolve(true)
            } else {
              requestPromiseCache.remove(key)
              if (requestErrorIsRetry && retryTimes < requestRetryMaxTimes) { // 请求执行后发生异常是否自动重试
                retryTimes++
                setTimeout(() => innerResolve(false), retryInterval)
              } else {
                innerResolve(true)
                reject(response)
              }
            }
          }))
        })
      } else {
        requestPromiseCache.get(key).then(() => {
          resolve(requestCache.get(key))
        })
      }
    }, { maxTimeMs: (Math.min(requestRetryMaxTimes, 5) + 1) * (Math.max(retryInterval, 2000) + 500) })
  })
}
